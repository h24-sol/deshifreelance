async function submitProof(taskId) { const proofInput = document.getElementById(`proof-${taskId}`); const proofText = proofInput.value.trim(); if (!proofText) { alert('Please enter your proof before submitting.'); return; } try { const res = await fetch(`https://deshifreelance.onrender.com/api/tasks/${taskId}/submit`, { method: 'POST', headers: { 'Content-Type': 'application/json', 'x-auth-token': localStorage.getItem('token') }, body: JSON.stringify({ proof: proofText }) }); const data = await res.json(); if (res.ok) { alert('Proof submitted successfully! Awaiting employer approval.'); proofInput.value = ''; } else { alert(data.message || 'Failed to submit proof.'); } } catch (err) { alert('Error submitting proof. Check console for details.'); } }
